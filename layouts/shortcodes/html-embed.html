{{- /* Local authored HTML stays in its own document; the theme owns the frame. */ -}} {{- $src := .Get "src" | default "" -}} {{- $title := .Get "title" | default "" -}} {{- $mode := .Get "mode" | default "page" -}} {{- if or (not $title) (not (findRE `^/?[^:/?#]+(?:/[^/?#]+)*\.html$` $src)) (in $src "..") (in $src "\\") -}} {{- errorf "html-embed requires title and a local .html src without query, fragment or traversal: %s" .Position -}} {{- end -}} {{- if not (in (slice "page" "archify") $mode) }}{{ errorf "html-embed mode must be page or archify: %s" .Position }}{{ end -}} {{- $surface := .Get "surface" | default (cond (eq $mode "archify") "theme" "source") -}} {{- if or (not (in (slice "theme" "source") $surface)) (and (eq $mode "page") (eq $surface "theme")) }}{{ errorf "html-embed surface must be source, or theme with mode=archify: %s" .Position }}{{ end -}} {{- $heightValue := "480" -}} {{- if isset .Params "height" }}{{ $heightValue = printf "%v" (.Get "height") }}{{ end -}} {{- $widthValue := printf "%v" (.Get "minWidth" | default 0) -}} {{- if or (not (findRE `^[0-9]+$` $heightValue)) (not (findRE `^[0-9]+$` $widthValue)) }}{{ errorf "html-embed dimensions must be integers: %s" .Position }}{{ end -}} {{- $height := int $heightValue -}} {{- $minWidth := int $widthValue -}} {{- if or (lt $height 160) (gt $height 1600) (lt $minWidth 0) (gt $minWidth 2400) }}{{ errorf "html-embed requires height 160..1600 and minWidth 0..2400: %s" .Position }}{{ end -}} {{- /* Preserve the width floor of existing minWidth-only calls. */ -}} {{- $fit := .Get "fit" | default (cond (gt $minWidth 0) "scroll" "width") -}} {{- if not (in (slice "width" "scroll") $fit) }}{{ errorf "html-embed fit must be width or scroll: %s" .Position }}{{ end -}} {{- $initialWidth := cond (eq $fit "width") 0 $minWidth -}} {{- $url := strings.TrimPrefix "/" $src | relURL -}} {{- $frameURL := $url -}} {{- if eq $mode "archify" }}{{ $frameURL = printf "%s?embed=1&theme=%s" $url (cond (eq (site.Params.defaultScheme | default "paper") "dark") "dark" "light") }}{{ end -}}
