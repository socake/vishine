{{- /* Read rendered headings, including headings emitted by shortcodes. IDs come from Hugo/render hooks; never manufacture a second set of anchors. */ -}} {{- $headings := slice -}} {{- range findRE `(?s)]*\bid="[^"]+"[^>]*>.*?` .Content -}} {{- $id := replaceRE `(?s)^]*\bid="([^"]+)".*$` "$1" . -}} {{- $level := int (replaceRE `(?s)^]*class="anchor".*?` "" . | plainify | htmlUnescape | strings.TrimSpace -}} {{- $headings = $headings | append (dict "id" $id "level" $level "title" $title) -}} {{- end -}} {{- return $headings -}}
